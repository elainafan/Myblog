void solve() {
    int n, m;
    cin >> n >> m;
    vector<pll> ma(n);
    rep(i, 0, n - 1) cin >> ma[i].first;
    rep(i, 0, n - 1) cin >> ma[i].second;
    map<ll, ll> ma2;
    for (auto& [x, r] : ma) {
        rep(i, x - r, x + r) {
            if (!ma2.count(i)) {
                ma2[i] = 2 * (int)sqrt(r * r - (x - i) * (x - i)) + 1;
            } else
                ma2[i] = max(ma2[i], 2LL * (int)sqrt(r * r - (x - i) * (x - i)) + 1);
        }
    }
    ll ans = 0;
    for (auto& [x, y] : ma2) ans += y;
    cout << ans << endl;
    return;
}

mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
int ask(int l, int r, int v) {
    int op;
    cout << "? " << l << ' ' << r << ' ' << v << '\n';
    cout.flush();
    cin >> op;
    return op;

void report(int l, int r, int v) {
    cout << "! " << l << ' ' << r << ' ' << v << '\n';
    cout.flush();
}
void solve() {
    int n;
    cin >> n;
    int x = 1, y = 2, z = 3;
    while (true) {
        int tem = rng() % 3;
        int op = ask(x, y, z);
        if (op == 0) {
            report(x, y, z);
            return ;
        }
        if (tem == 0)
            x = op;
        else if (tem == 1)
            y = op;
        else
            z = op;
    }
    return;
}

void solve() {
    ll a, b, c, d;
    cin >> a >> b >> c >> d;
    auto check = [&](ll l, ll r) -> vl {
        vl cnt(62);
        while (l < r) {
            ll tem;
            if (l == 0) {
                tem = 63 - __builtin_clzll(r);
                tem = (1LL << tem);
            } else {
                tem = lowbit(l);
                while (tem > r - l) tem >>= 1;
            }
            cnt[__builtin_ctzll(tem)]++;
            l += tem;
        }
        return cnt;
    };
    vl cnt1 = check(a, b);
    vl cnt2 = check(c, d);
    ll ans = 0;
    rep(i, 0, 61) {
        rep(j, 0, 61) { ans += cnt1[i] * cnt2[j] * (1LL << (abs(j - i))); }
    }

    return;
}

void solve() {
    int n;
    cin >> n;
    vl a(n);
    rep(i, 0, n - 1) cin >> a[i];
    rep(i, 0, n - 1) a.push_back(a[i]);
    vvl dp(2 * n, vl(2 * n));
    rep(i, 3, n) {
        rep(j, 0, 2 * n - i) {
            rep(k, j, j + i - 2) dp[j][j + i - 1] = max(dp[j][j + i - 1], dp[j][k] + dp[k + 1][j + i - 1]);
            rep(k, j + 1, j + i - 2) dp[j][j + i - 1] =
                max(dp[j][j + i - 1], dp[j + 1][k - 1] + dp[k + 1][j + i - 2] + a[j] * a[k] * a[j + i - 1]);
        }
    }
    ll ans = LLONG_MIN;
    rep(i, 0, n - 1) ans = max(ans, dp[i][i + n - 1]);
    cout << ans << endl;
    return;
}
