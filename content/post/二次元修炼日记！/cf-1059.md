void report(ll a, ll b) {
    cout << "! " << a << ' ' << b << '\n';
    cout.flush();
}
void solve() {
    ll n;
    cin >> n;
    ll tem = ask(2, 1, n) - ask(1, 1, n);
    ll l = 1, r = n, ans = 1, mid;
    auto check = [&](ll mid) -> bool { return ask(2, 1, mid) - ask(1, 1, mid) >= tem; };
    while (l <= r) {
        mid = (l + r) / 2;
        if (check(mid)) {
            ans = mid;
            r = mid - 1;
        } else
            l = mid + 1;
    }
    report(ans - tem + 1, ans);
    return;
}

void solve() {

        maxx = min(maxx, ma[i].second);

        res[mixx] = 0;
        int idx = 1;

            res[i] = idx++;
        }
        rep(i, 0, n - 1) cout << res[i] << ' ';

        return;

                flag = true;
            }
        }

        res[idx] = 0;
        res[idx + 1] = 1;

void solve() {

        return ;
    }
    vvi ma(n + 1);
    ma[1].push_back(2);
    ma[1].push_back(5);
    ma[2].push_back(3);
    ma[3].push_back(4);

        ma[1].pop_back();
        ma[2].push_back(i - 1);
        ma[1].push_back(i);
    }

        for (auto& p : ma[i]) cout << i << ' ' << p << endl;
    }
    return;
}
