#include <iostream>
using namespace std;

class Midterm {
private:
    int val;

public:
    friend ostream &operator<<(ostream &os, const Midterm &t) {
        os << t.val;
        return os;
    }
    Midterm operator+(int x) {
        val += x;
        Midterm temp;
        temp.val = val;
        return temp;
    }
    Midterm(int x) : val(x) {}
    Midterm() {}
    Midterm operator++(int) {
        Midterm temp;
        val++;
        temp.val = val;
        return temp;
    }
    operator int() { return val; }
    Midterm &operator-=(int x) {
        val -= x;
        return *this;
    }
    // 在此处补充你的代码
};

        return a;

        return b;

class MyFunc {

        return;

        }

class Rectangle {
public:
    int l, w;
    int c, s;
    Rectangle(int a, int b) : l(a), w(b) {
        c = 2 * (a + b);
        s = a * b;
    }
    friend ostream &operator<<(ostream &os, const Rectangle &t) {
        os << t.s << ' ' << t.c;
        return os;

    bool operator<(const Rectangle &k) const {
        if (s == k.s) return c > k.c;

            cnt += a;
            tot += a * b;
        }
