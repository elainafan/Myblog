                it = s.erase(it);

            if (!flag) {

                flag = true;
            } else {
                if ((a[i] - a[idx]) != x * (i - idx)) {
                    flag = 0;
                    break;
                }
            }
        }
        if (!flag)
            return false;
        else
            return true;
    };
    if (check(tem1) || check(tem2) || check(tem3)) {
        cout << "YES" << endl;
        return;
    }
    cout << "NO" << endl;
