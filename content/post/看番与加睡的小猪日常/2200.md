void solve() {
    int n, m;
    int x, y;
    cin >> n >> m;
    vvi ma(n);
    vi deg(n);
    rep(i, 0, m - 1) {
        cin >> x >> y;
        ma[x - 1].push_back(y - 1);
        deg[y - 1]++;
    }
    queue<int> q;
    auto check = [&](int x) -> bool {
        auto deg2 = deg;
        deg2[x] = max(deg2[x] - 1, 0);
        vi res;
        while (!q.empty()) q.pop();
        rep(i, 0, n - 1) {
            if (!deg2[i]) q.push(i);
        }
        while (!q.empty()) {
            auto node = q.front();
            q.pop();
            res.push_back(node);
            for (int& p : ma[node]) {
                if (--deg2[p] == 0) q.push(p);
            }
        }
        return sz(res) == n;
    };
    rep(i, 0, n - 1) {
        if (check(i)) {
            cout << "YES" << endl;
            return;
        }
    }
    cout << "NO" << endl;
    return;
}
