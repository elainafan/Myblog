void solve() {
    int n, q;
    int l, r;
    cin >> n >> q;
    vi a(n);
    rep(i, 0, n - 1) cin >> a[i];
    vl pre(n);
    vector<bool> vis(n, false);
    rep(i, 1, n - 2) {
        if (a[i] <= a[i - 1] && a[i + 1] <= a[i]) vis[i] = true;
    }
    rep(i, 1, n - 1) { pre[i] = pre[i - 1] + vis[i]; }
    rep(i, 0, q - 1) {
        cin >> l >> r;
        l--, r--;
        if (r - l + 1 <= 2) {
            cout << r - l + 1 << endl;
            continue;
        }
        int tem = (r - l + 1) - (pre[r - 1] - pre[l]);
        cout << tem << endl;
    }
    return;
}
