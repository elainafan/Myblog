int ask(int l, int r) {
    cout << "? " << l << ' ' << r << '\n';
    cout.flush();
    int op;
    cin >> op;
    return op;
}
void report(ll sum, ll sum2) {
    cout << "! " << sum << ' ' << sum2 << '\n';
    cout.flush();
    int op;
    cin >> op;
    if (op == 1)
        return;
    else
        exit(0);
}
void solve() {
    int n;
    cin >> n;
    pii ans = make_pair(1, 2);
    auto upd = [&](int x) -> void {
        int op1 = ask(ans.first, x);
        int op2 = ask(ans.second, x);
        if (op1 == op2)
            ans = make_pair(ans.first, ans.second);
        else if (op1 > op2)
            ans = make_pair(ans.first, x);
        else
            ans = make_pair(ans.second, x);
    };
    rep(i, 3, n) { upd(i); }
    report(ans.first, ans.second);
    return;
}
