```cpp
void solve() {
    ll n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vl a(n + 1, 1);
    int tem = 0;
    int l = 1, r = n;
    rep(i, 0, k - 1) {
        if (s[i] == '0') {
            a[l] = 0;
            l++;
        } else if (s[i] == '1') {
            a[r] = 0;
            r--;
        } else
            tem++;
    }
    if (r - l + 1 <= tem) {
        rep(i, 0, n - 1) cout << '-';
        cout << endl;
        return ;
    }
    rep(i, l, l + tem - 1) {
        if (a[i] == 0)
            continue;
        else
            a[i] = 2;
    }
    frep(i, r, r - tem + 1) {
        if (a[i] == 0)
            continue;
        else
            a[i] = 2;
    }
    rep(i, 1, n) {
        if (a[i] == 0)
            cout << '-';
        else if (a[i] == 1)
            cout << '+';
        else
            cout << '?';
    }
    cout << endl;
    return;
}
```

```cpp
void solve() {
    ll n;
    cin >> n;
    string s;
    cin >> s;
    int tot0 = 0, tot1 = 0;
    rep(i, 0, n - 1) {
        if (s[i] == 'a')
            tot0++;
        else
            tot1++;
    }
    map<ll, ll> ma;
    ma[0] = -1;
    ll tem = tot0 - tot1;
    if (tem == 0) {
        cout << 0 << endl;
        return;
    }
    ll tem2 = 0;
    ll ans = LLONG_MAX;
    rep(i, 0, n - 1) {
        tem2 += (s[i] == 'a' ? 1 : -1);
        if (ma.find(tem2 - tem) != ma.end()) ans = min(ans, i - ma[tem2 - tem]);
        ma[tem2] = i;
    }
    cout << (ans == n ? -1 : ans) << endl;
    return;
}
```
