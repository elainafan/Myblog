## B
题目大意：给定目标坐标和一串操作类型，判断这些操作能否走到目标。

数据范围：见原题。

思路：把坐标取绝对值后，只需要关心两类操作数量。一步操作最多补一维或两维，判断步数和曼哈顿距离上下界即可。

```cpp
void solve() {
    int n;
    ll x, y;
    cin >> n >> x >> y;
    x = abs(x), y = abs(y);
    string s;
    cin >> s;
    ll tem1 = 0, tem2 = 0;
    ll res1 = x + y, res2 = x - y;
    int tot1 = 0, tot2 = 0;
    rep(i, 0, n - 1) {
        if (s[i] == '4') {
            tot1++;
        } else {
            tot2++;
        }
    }
    if (tot1 + tot2 >= max(x, y) && x + y <= tot1 + 2 * tot2) {
        cout << "YES" << endl;
        return;
    }
    cout << "NO" << endl;
    return;
}
```

## C
题目大意：根据区间约束构造一个数组，使不同类型的区间满足 mex 相关要求。

数据范围：见原题。

思路：对每个位置记录覆盖它的约束类型。只被类型 1 覆盖的位置填 $k$，只被类型 2 覆盖的位置循环填 $0\ldots k-1$，其他位置填很大的数。

```cpp
void solve() {
    int n, k, q;
    int c, l, r;
    cin >> n >> k >> q;
    vector<set<int>> ma(n);
    vi a(n, 1e9);
    rep(i, 0, q - 1) {
        cin >> c >> l >> r;
        l--, r--;
        rep(j, l, r) ma[j].insert(c);
    }
    int cnt = 0;
    rep(i, 0, n - 1) {
        if (ma[i].count(1) && !ma[i].count(2)) {
            a[i] = k;
        }
        if (!ma[i].count(1) && ma[i].count(2)) {
            a[i] = cnt;
            cnt = (cnt + 1) % k;
        }
    }
    rep(i, 0, n - 1) cout << a[i] << ' ';
    cout << endl;
    return;
}
```

## D
题目大意：给定数组和区间 $[l,r]$，把每个数调整到某个目标附近，求最小总调整代价。

数据范围：见原题。

思路：先把所有数夹到 $[l,r]$ 里，超出的部分直接贡献代价。剩下的数排序后首尾配对，总代价就是每对较大值减较小值。

```cpp
void solve() {
    int n, l, r;
    cin >> n >> l >> r;
    vl a(n);
    rep(i, 0, n - 1) cin >> a[i];
    ranges::sort(a);
    vl pre(n);
    vl suf(n + 1);
    pre[0] = a[0];
    suf[n - 1] = a[n - 1];
    rep(i, 1, n - 1) pre[i] = pre[i - 1] + a[i];
    frep(i, n - 2, 0) suf[i] = suf[i + 1] + a[i];
    ll ans = 0;
    // 枚举选择a[i]-p的数量
    // 假设a[i]-p的数量为y，p-a[i]的数量为x
    // 当y>=x时，p应该取l
    // 当y<x时，p应该取r
    rep(i, 0, n - 1) {
        if (a[i] < l) ans += l - a[i], a[i] = l;
        if (a[i] > r) ans += a[i] - r, a[i] = r;
    }
    for (int i = 0, j = n - 1; i < j; i++, j--) {
        ans += a[j] - a[i];
    }
    cout << ans << endl;
    return;
}
```
