void solve() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vvi dp(n + 1, vi(n + 1));
    vvi pre(n + 1, vi(n + 1));
    vvi lcp(n + 2, vi(n + 2));
    frep(i, n, 1) {
        frep(j, n, 1) {
            if (s[i - 1] == s[j - 1]) lcp[i][j] = lcp[i + 1][j + 1] + 1;
        }
    }
    rep(i, 1, n) dp[i][i] = 1;
    rep(i, 1, n) {
        rep(j, 1, i) {
            if (j < i) {
                if (i - 2 * j >= 0 && s[i - j] != '0') {
                    int l = lcp[i - 2 * j + 1][i - j + 1];
                    if (l < j && s[i - 2 * j + l] < s[i - j + l]) dp[i][j] = (dp[i][j] + dp[i - j][j]) % MOD;
                }
                if (s[i - j] != '0') dp[i][j] = (dp[i][j] + pre[i - j][j - 1]) % MOD;
            }
            pre[i][j] = (pre[i][j - 1] + dp[i][j]) % MOD;
        }
        rep(j, i + 1, n) pre[i][j] = pre[i][i];
    }
    ll ans = 0;
    rep(i, 1, n) {
        ans += dp[n][i] % MOD;
        ans %= MOD;
    }
    cout << ans << endl;
    return;
}
