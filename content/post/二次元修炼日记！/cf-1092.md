    ll ans = 0;
    ll tem = min(a, c);
    a -= tem, c -= tem, ans += 4 * tem;
    if (a > 0 && b > 0) {
        if (a <= 2 * b)
            ans += 3 * b + 2 * a;
        else
            ans += 3 * b + 2 * a + 1;
    } else if (a > 0) {
        ans += 2 * a + 1;
    } else if (b > 0) {
        ans += 3 * b;
    }
    if (c > 0) ans += 3 * c;
    cout << ans << endl;
    return;
}
