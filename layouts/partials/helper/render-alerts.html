{{/* Hugo 0.131 emits GitHub alert markers as the first paragraph of a blockquote. Change only its opening; Goldmark has already rendered the rest of the block. */}} {{- $html := . -}} {{- if in $html "[!" -}} {{- range $kind, $label := dict "NOTE" "说明" "TIP" "提示" "IMPORTANT" "重要" "WARNING" "警告" "CAUTION" "注意" -}} {{- $marker := printf `(?i)
\s*

\[!%s\][ \t]*` $kind -}} {{- $title := printf `

%s

` (lower $kind) $label -}} {{- $html = replaceRE (print $marker "

") $title $html -}} {{- $html = replaceRE (print $marker `(?:\s*|\r?\n)`) (print $title "\n

") $html -}} {{- end -}} {{- end -}} {{- return ($html | safeHTML) -}}
